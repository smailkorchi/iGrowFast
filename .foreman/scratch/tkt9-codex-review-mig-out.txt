-- TKT-9: package status history — our own persistent record of every parcel
-- status transition (L1, append-only, trigger-captured) plus the last rich
-- provider-history snapshot (L2, upserted by provider-integration track_parcel).
-- Resilience goal: tracking renders from OUR data when the provider API is down.
--
-- Scale notes (measured 2026-07-29): ~120 transitions/day today; at ×100 growth
-- ≈ 4.4M rows/yr ≈ 0.5 GB/yr — unpartitioned is fine for years. Revisit range
-- partitioning by created_at only if observed volume exceeds ~10M rows/yr.
-- Retention: FOREVER (history is the point). No realtime publication ever.

BEGIN;

-- ── L1: transition log ─────────────────────────────────────────────────────
CREATE TABLE public.package_status_history (
    id                 uuid PRIMARY KEY DEFAULT gen_random_uuid(),
    "DigiCom Business" uuid NOT NULL REFERENCES public.businesses(id) ON DELETE CASCADE,
    order_id           uuid REFERENCES public.orders(id) ON DELETE CASCADE,
    -- Parcel FKs are SET NULL on purpose: parcels get deleted (delivery-note
    -- flows) but their history must survive — tracking_code keeps the link.
    company_parcel_id  uuid REFERENCES public.company_parcels(id) ON DELETE SET NULL,
    courier_parcel_id  uuid REFERENCES public.courier_parcels(id) ON DELETE SET NULL,
    tracking_code      text,
    old_status_id      uuid REFERENCES public.package_statuses(id) ON DELETE SET NULL,
    new_status_id      uuid NOT NULL REFERENCES public.package_statuses(id) ON DELETE RESTRICT,
    created_at         timestamptz NOT NULL DEFAULT now()
);

CREATE INDEX idx_psh_tracking_created ON public.package_status_history (tracking_code, created_at);
CREATE INDEX idx_psh_order_created ON public.package_status_history (order_id, created_at);
CREATE INDEX idx_psh_biz_created ON public.package_status_history ("DigiCom Business", created_at DESC);

ALTER TABLE public.package_status_history ENABLE ROW LEVEL SECURITY;

-- Tenant read-only; the SECURITY DEFINER trigger below and the service role are
-- the only writers (no INSERT/UPDATE/DELETE policies for users).
CREATE POLICY psh_tenant_select ON public.package_status_history
    FOR SELECT TO authenticated
    USING ("DigiCom Business" = (SELECT current_user_digicom_business_id()));

GRANT SELECT ON public.package_status_history TO authenticated;

-- ── Trigger: capture every status transition on both parcel tables ─────────
-- House pattern (auto_log_order_status_change): SECDEF + IS DISTINCT FROM
-- guard. Single INSERT from NEW/OLD only — no lookups on the hot path.
CREATE OR REPLACE FUNCTION public.log_package_status_change()
RETURNS trigger
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path TO 'public', 'pg_catalog'
AS $$
BEGIN
    IF NEW.status_id IS NULL THEN
        RETURN NEW;
    END IF;
    IF TG_OP = 'UPDATE' AND NEW.status_id IS NOT DISTINCT FROM OLD.status_id THEN
        RETURN NEW;
    END IF;

    INSERT INTO public.package_status_history (
        "DigiCom Business", order_id, company_parcel_id, courier_parcel_id,
        tracking_code, old_status_id, new_status_id
    ) VALUES (
        NEW."DigiCom Business",
        NEW.order_id,
        CASE WHEN TG_TABLE_NAME = 'company_parcels' THEN NEW.id END,
        CASE WHEN TG_TABLE_NAME = 'courier_parcels' THEN NEW.id END,
        NULLIF(btrim(coalesce(NEW.tracking_code, '')), ''),
        CASE WHEN TG_OP = 'UPDATE' THEN OLD.status_id END,
        NEW.status_id
    );
    RETURN NEW;
END;
$$;

CREATE TRIGGER trg_log_company_parcel_status_history
    AFTER INSERT OR UPDATE OF status_id ON public.company_parcels
    FOR EACH ROW EXECUTE FUNCTION public.log_package_status_change();

CREATE TRIGGER trg_log_courier_parcel_status_history
    AFTER INSERT OR UPDATE OF status_id ON public.courier_parcels
    FOR EACH ROW EXECUTE FUNCTION public.log_package_status_change();

-- ── Backfill: one baseline row per existing parcel at its current status ───
-- created_at = the parcel's updated_at (the real "as of" moment we know).
INSERT INTO public.package_status_history
    ("DigiCom Business", order_id, company_parcel_id, tracking_code, new_status_id, created_at)
SELECT cp."DigiCom Business", cp.order_id, cp.id,
       NULLIF(btrim(coalesce(cp.tracking_code, '')), ''),
       cp.status_id, coalesce(cp.updated_at, cp.created_at, now())
FROM public.company_parcels cp
WHERE cp.status_id IS NOT NULL;

INSERT INTO public.package_status_history
    ("DigiCom Business", order_id, courier_parcel_id, tracking_code, new_status_id, created_at)
SELECT crp."DigiCom Business", crp.order_id, crp.id,
       NULLIF(btrim(coalesce(crp.tracking_code, '')), ''),
       crp.status_id, coalesce(crp.updated_at, crp.created_at, now())
FROM public.courier_parcels crp
WHERE crp.status_id IS NOT NULL;

-- ── L2: last successful rich provider-history snapshot ─────────────────────
CREATE TABLE public.parcel_tracking_snapshots (
    provider_id        uuid NOT NULL REFERENCES public.providers(id) ON DELETE CASCADE,
    tracking_code      text NOT NULL,
    "DigiCom Business" uuid NOT NULL REFERENCES public.businesses(id) ON DELETE CASCADE,
    order_id           uuid REFERENCES public.orders(id) ON DELETE CASCADE,
    status             text,
    status_label       text,
    history            jsonb NOT NULL DEFAULT '[]'::jsonb,
    fetched_at         timestamptz NOT NULL DEFAULT now(),
    PRIMARY KEY (provider_id, tracking_code)
);

ALTER TABLE public.parcel_tracking_snapshots ENABLE ROW LEVEL SECURITY;

CREATE POLICY pts_tenant_select ON public.parcel_tracking_snapshots
    FOR SELECT TO authenticated
    USING ("DigiCom Business" = (SELECT current_user_digicom_business_id()));

GRANT SELECT ON public.parcel_tracking_snapshots TO authenticated;

COMMIT;
# TKT-9 — Package status history (design, LEAD, 2026-07-29)

## Goal
Own the package-status timeline: if the provider API is down (or purges old history), tracking still renders from OUR data. Clean, scalable to thousands of users, frugal on the box (PostgREST pool ≈10, no realtime amplification).

## Two layers

### L1 — `package_status_history` (append-only transition log; canonical)
One row per status transition of a parcel, captured by a DB TRIGGER on `company_parcels` + `courier_parcels` (`AFTER INSERT OR UPDATE OF status_id`). Trigger = catches EVERY writer (webhook, pollers via webhook, TKT-6 retro-apply, batch ops, manual SQL) with zero app-code dependence. Mirrors the `auto_log_order_status_change` house pattern (SECDEF fn, IS DISTINCT FROM guard) minus the UI-dedup window (no UI writes here — trigger is the only writer).
- Columns: id, "DigiCom Business" (CASCADE), order_id (CASCADE), company_parcel_id (SET NULL), courier_parcel_id (SET NULL), tracking_code text (denormalized — history survives parcel deletion), old_status_id (SET NULL), new_status_id (RESTRICT), created_at.
- Parcel FKs SET NULL deliberately: parcels get deleted (delivery-note flows); history must survive. order_id CASCADE: hard-deleted order → its history is meaningless.
- Indexes (3): (tracking_code, created_at), (order_id, created_at), ("DigiCom Business", created_at DESC). No status_id indexes — never queried by status.
- RLS: tenant SELECT only (authenticated); no INSERT/UPDATE/DELETE policies — SECDEF trigger + service role write.
- NO realtime publication, NO broadcast triggers.
- Retention: FOREVER — history is the point. Row ≈ 120 B. Scale math: today ~120 webhook transitions/day (826/7d, 1,988 parcels); at ×100 growth ≈ 12k rows/day ≈ 4.4M/yr ≈ 0.5 GB/yr. Unpartitioned is fine for years; partition by created_at range ONLY if volume observed >10M/yr (noted in migration comment; schema is partition-friendly: no inbound FKs, time-clustered writes).
- Backfill: one seed row per existing parcel at its current status (created_at = parcel.updated_at, old=NULL) — every parcel has a baseline; forward history grows from deploy.

### L2 — `parcel_tracking_snapshots` (rich provider-history cache)
The provider timeline carries data we can never reconstruct (comments « Facture … CRBT: 165 », livreur name/phone). Persist the last successful fetch:
- PK (provider_id, tracking_code); columns: "DigiCom Business" (CASCADE), order_id (CASCADE), status, status_label, history jsonb (the adapter's normalized array), fetched_at.
- Written by provider-integration `track_parcel` success path (service role, FAIL-OPEN, one reverse lookup of company_parcels by provider_id+tracking_code for biz/order). Write rate = user-driven tracking opens, dampened by the existing 10-min client cache — trivially low.
- One row per parcel ever tracked (upsert). Size ≈ #parcels × ~4 KB.
- RLS: tenant SELECT only.

### Read/fallback path (ParcelTrackingDialog)
Provider fetch fails (today: toast + close) → NEW: (1) query snapshot by tracking_code → render FULL history + quiet banner « Historique fournisseur indisponible — données du {fetched_at} » ; (2) else query L1 by tracking_code → slim timeline (label/color/icon via resolveById(new_status_id), date) + same banner; (3) else current behavior. Fallback feeds the SAME normalized trackingHistory state, so the TKT-8b demand cards and all downstream rendering work unchanged.

## Deliberately OUT (documented)
- No trigger on orders.package_status_id: it mirrors parcel status (sync triggers) → would double-log; orders without parcels get no provider webhooks (webhook resolves order THROUGH the parcel), so nothing is missed. Manual admin edits of the order mirror are not provider history.
- No pruning cron. No partitioning yet. No MB policies (no parcel UI for MBs).

## Execution stages
1. Migration authored → **Codex review BEFORE applying** → apply → BEGIN/ROLLBACK trigger simulations on box (zero side effects) → backfill verify counts.
2. Edge worker (snapshot upsert) ∥ frontend worker (fallback UI) → blind verifiers → Codex diff review.
3. Deploy edge + frontend; live checks: track a real parcel → snapshot row exists; synthetic-history render verified by code review (no logged-in browser available).

CREATE OR REPLACE FUNCTION public.clear_order_logistics_when_no_parcels()
 RETURNS trigger LANGUAGE plpgsql SECURITY DEFINER SET search_path TO 'public'
AS $function$
DECLARE
  v_confirmed_id uuid;
  v_processed_id uuid;
BEGIN
  IF OLD.order_id IS NULL OR NOT EXISTS (SELECT 1 FROM public.orders WHERE id = OLD.order_id) THEN
    RETURN OLD;
  END IF;
  IF TG_TABLE_NAME = 'company_parcels' THEN
    IF OLD.company_delivery_note_id IS NULL THEN RETURN OLD; END IF;
  ELSIF TG_TABLE_NAME = 'courier_parcels' THEN
    IF OLD.courier_delivery_note_id IS NULL THEN RETURN OLD; END IF;
  END IF;
  IF EXISTS (SELECT 1 FROM public.company_parcels WHERE order_id = OLD.order_id)
     OR EXISTS (SELECT 1 FROM public.courier_parcels WHERE order_id = OLD.order_id) THEN
    RETURN OLD;
  END IF;
  SELECT id INTO v_confirmed_id FROM public.order_statuses WHERE code = 'CONFIRMED' LIMIT 1;
  SELECT id INTO v_processed_id FROM public.order_statuses WHERE code = 'PROCESSED' LIMIT 1;
  UPDATE public.orders
  SET tracking_number          = NULL,
      package_status_id        = NULL,
      current_delivery_note_id = NULL,
      processing               = false,
      courier_id               = NULL,
      order_status_id          = CASE
                                   WHEN order_status_id = v_processed_id AND v_confirmed_id IS NOT NULL
                                   THEN v_confirmed_id ELSE order_status_id END
  WHERE id = OLD.order_id;
  RETURN OLD;
END;
$function$;

-- BUG (T7 trigger-sweep #1, 2026-06-29): courier unpaid-on-delivery trigger over-fires and can
-- silently revert a delivered courier order back to UNPAID on an unrelated parcel edit.
--
-- trigger_set_order_unpaid_on_delivery_courier was AFTER INSERT OR UPDATE (ALL columns, no scope),
-- while its company twin trigger_set_order_unpaid_on_delivery is scoped to UPDATE OF status_id.
-- The shared function set_order_unpaid_on_delivery() had NO status-change guard (unlike its siblings
-- set_courier_parcel_delivered_at / sync_courier_parcel_status_to_order, which both short-circuit on
-- `NEW.status_id IS NOT DISTINCT FROM OLD.status_id`). So ANY courier_parcels UPDATE (tracking_code,
-- is_fragile, is_open, is_try_on, delivered_at, ...) re-ran the function; if the parcel's current
-- status is_delivered=true, it re-asserted orders.order_status_id = UNPAID. The in-function UPDATE
-- already protects paid / returned / replace / return-note / already-unpaid orders, but a DELIVERED
-- order parked in any other non-paid limbo status (cancelled-post-delivery, suivi, etc.) would be
-- flipped to UNPAID by an edit that never touched status. Latent today (0 orders in the vulnerable
-- state) but a real money-column footgun.
--
-- FIX (two layers):
--  1) Add the same change-guard the sibling functions carry, at the top of the shared function. This
--     makes BOTH company and courier paths no-op on any non-status update (and on a redundant
--     `SET status_id = status_id`). This is the robust core fix.
--  2) Scope the courier trigger to `AFTER INSERT OR UPDATE OF status_id`, mirroring the company twin
--     exactly (the company unpaid twin has NO WHEN clause; order_id is guarded inside the function),
--     so the function is not even entered on unrelated courier_parcels edits.
--
-- Behavior on a genuine delivery transition is unchanged. Rollback: drop the guard and recreate the
-- courier trigger AFTER INSERT OR UPDATE (all columns).

CREATE OR REPLACE FUNCTION public.set_order_unpaid_on_delivery()
 RETURNS trigger
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public', 'pg_catalog'
AS $function$
DECLARE
    unpaid_status_id uuid;
    v_is_delivered boolean := false;
BEGIN
    -- Only act on an actual status change (mirrors set_courier_parcel_delivered_at /
    -- sync_courier_parcel_status_to_order). Prevents unrelated parcel-attribute edits from
    -- re-asserting UNPAID on an already-delivered order.
    IF TG_OP = 'UPDATE' AND NEW.status_id IS NOT DISTINCT FROM OLD.status_id THEN
        RETURN NEW;
    END IF;

    IF NEW.status_id IS NOT NULL THEN
        SELECT is_delivered INTO v_is_delivered FROM public.package_statuses WHERE id = NEW.status_id;
        v_is_delivered := COALESCE(v_is_delivered, false);
    END IF;

    IF v_is_delivered AND NEW.order_id IS NOT NULL THEN
        SELECT id INTO unpaid_status_id
        FROM public.order_statuses
        WHERE is_unpaid AND (active IS DISTINCT FROM false)
        LIMIT 1;

        IF unpaid_status_id IS NOT NULL THEN
            UPDATE public.orders o
            SET order_status_id = unpaid_status_id
            FROM public.order_statuses cur
            WHERE o.id = NEW.order_id
              AND o.order_status_id IS DISTINCT FROM unpaid_status_id
              AND cur.id = o.order_status_id
              AND NOT cur.is_paid
              AND NOT cur.is_return
              AND o.current_return_note_id IS NULL
              AND o.current_courier_return_note_id IS NULL
              AND o.replace IS NOT TRUE;
        END IF;
    END IF;

    RETURN NEW;
END;
$function$;

-- ── orders: coalesced statement events + capped pkg toasts ───────────────────
CREATE OR REPLACE FUNCTION public.broadcast_orders_stmt()
RETURNS trigger
LANGUAGE plpgsql SECURITY DEFINER
SET search_path TO ''
AS $function$
declare
  r record;
  t record;
  v_pkg_changes int;
  v_tracking text; v_pname text; v_ptype text; v_plogo text;
begin
  begin
    if tg_op = 'INSERT' then
      for r in
        select n."DigiCom Business" as biz,
               count(*) as cnt,
               bool_or(n.source_review_status = 'pending') as review_pending,
               case when count(distinct public.normalize_phone_for_customer(n.phone)) = 1
                    then min(public.normalize_phone_for_customer(n.phone)) end as phone
        from new_rows n
        where n."DigiCom Business" is not null
        group by n."DigiCom Business"
      loop
        perform realtime.send(
          jsonb_strip_nulls(jsonb_build_object(
            'op', 'INSERT', 'count', r.cnt, 'phone', r.phone,
            'review_pending', case when r.review_pending then true end
          )),
          'order', 'ecom:biz:' || r.biz::text, true);
      end loop;

    elsif tg_op = 'UPDATE' then
      for r in
        select n."DigiCom Business" as biz,
               count(*) as cnt,
               case when count(distinct public.normalize_phone_for_customer(n.phone)) = 1
                    then min(public.normalize_phone_for_customer(n.phone)) end as phone
        from new_rows n
        where n."DigiCom Business" is not null
        group by n."DigiCom Business"
      loop
        perform realtime.send(
          jsonb_strip_nulls(jsonb_build_object('op', 'UPDATE', 'count', r.cnt, 'phone', r.phone)),
          'order', 'ecom:biz:' || r.biz::text, true);
      end loop;

      -- pkg-toast rows: package_status_id transitions, per-row parity payload,
      -- capped at 10 per statement (bulk flips invalidate via the coalesced
      -- event; they must not fire a toast per row).
      select count(*) into v_pkg_changes
      from new_rows n join old_rows o on o.id = n.id
      where n."DigiCom Business" is not null
        and o.package_status_id is distinct from n.package_status_id;

      if v_pkg_changes between 1 and 10 then
        for t in
          select n.*
          from new_rows n join old_rows o on o.id = n.id
          where n."DigiCom Business" is not null
            and o.package_status_id is distinct from n.package_status_id
        loop
          begin
            select cp.tracking_code, p.name, p.provider_type, p.logo_url
              into v_tracking, v_pname, v_ptype, v_plogo
              from public.company_parcels cp
              left join public.providers p on p.id = cp.provider_id
             where cp.order_id = t.id
             order by cp.created_at desc
             limit 1;
          exception when others then
            v_tracking := null; v_pname := null; v_ptype := null; v_plogo := null;
          end;
          perform realtime.send(
            jsonb_build_object(
              'op', 'UPDATE',
              'id', t.id::text,
              'phone', public.normalize_phone_for_customer(t.phone),
              'pkg', jsonb_build_object(
                'package_status_id', t.package_status_id::text,
                'name',              t.name,
                'seller_id',         t.seller_id::text,
                'tracking_code',     v_tracking,
                'provider_name',     v_pname,
                'provider_type',     v_ptype,
                'provider_logo_url', v_plogo)
            ) || case when t.date_deleted is not null
                      then jsonb_build_object('deleted', true) else '{}'::jsonb end,
            'order', 'ecom:biz:' || (t."DigiCom Business")::text, true);
        end loop;
      end if;

    elsif tg_op = 'DELETE' then
      for r in
        select o."DigiCom Business" as biz,
               count(*) as cnt,
               case when count(distinct public.normalize_phone_for_customer(o.phone)) = 1
                    then min(public.normalize_phone_for_customer(o.phone)) end as phone
        from old_rows o
        where o."DigiCom Business" is not null
        group by o."DigiCom Business"
      loop
        perform realtime.send(
          jsonb_strip_nulls(jsonb_build_object('op', 'DELETE', 'count', r.cnt, 'phone', r.phone)),
          'order', 'ecom:biz:' || r.biz::text, true);
      end loop;
    end if;
  exception when others then
    null;  -- never roll back the order write
  end;
  return null;
end;
$function$;

const demandText = (v: unknown): string => {
    if (v === null || v === undefined) return '';
    if (typeof v === 'boolean') return v ? 'Oui' : 'Non';

/** Only the fields that actually changed — skip equal, both-empty, or absent-in-new. */
const demandDiffs = (d: ChangeDemand) => {
    const oldV = (d.old_values || {}) as Record<string, unknown>;
    const newV = (d.new_values || {}) as Record<string, unknown>;
    return DEMAND_FIELDS.flatMap(({ key, label }) => {
        if (newV[key] === undefined) return [];
        const next = demandText(newV[key]);
        const prev = demandText(oldV[key]);
        if (!next && !prev) return [];
        // Numeric equality for the price only — a phone like 0612… must never be
        // compared as a number (leading zero would make it equal to 612…).
        const numeric = key === 'price' && prev !== '' && next !== ''
            && Number.isFinite(Number(prev)) && Number.isFinite(Number(next));
        if (numeric ? Number(prev) === Number(next) : prev === next) return [];
        return [{ key, label, prev, next }];
    });
};

-- ============================================================================
-- RLS tenant-scoping: status pipeline + users/businesses PII  (Security batch 1)
-- ----------------------------------------------------------------------------
-- Closes C1/C2 from the 2026-06 pipeline audit:
--   * Any authenticated user of ANY tenant could read/write EVERY tenant's
--     orders, company_parcels, courier_parcels, webhook_logs and
--     provider_webhooks (the last exposing webhook_secret -> webhook forgery).
--   * public.users SELECT ("Profiles are viewable by everyone") and
--     public.businesses SELECT ("Allow authenticated read") leaked cross-tenant
--     PII (bank RIB, CIN, salary, ICE/RC fiscal IDs) to any authenticated user
--     (users was readable by anon too).
--
-- Model: strict per-tenant via public.current_user_digicom_business_id().
-- Owners (role='owner', "DigiCom Business" IS NULL until they pick a business)
-- intentionally see nothing on the 5 status tables until selection (confirmed
-- with product owner). Owners retain cross-tenant visibility ONLY where the
-- product needs it: the business picker and owner consoles (is_owner()).
--
-- Verified on prod before authoring (2026-06-02):
--   * service_role.rolbypassrls = true  -> ALL edge-function ingestion
--     (provider-webhook, *-poll-webhook, provider-integration, ai-review-act)
--     bypasses RLS and is UNAFFECTED by these TO authenticated policies.
--   * authenticated / anon do NOT bypass RLS -> these policies govern them.
--   * 0 NULL "DigiCom Business" rows on orders / company_parcels /
--     courier_parcels / webhook_logs / provider_webhooks / businesses
--     -> equality predicate orphans no legitimate row.
--   * Helpers re-declared CREATE OR REPLACE (identical to live bodies) so the
--     migration is self-contained and applies cleanly on a fresh branch.
--
-- Scope guard: this migration ONLY swaps the OPEN SELECT / ALL visibility
-- policies. The already role-gated WRITE policies on users (check_user_role)
-- and businesses (creator_id / is_owner) are LEFT INTACT.
-- NOTE: webhook_secret column-level REVOKE is intentionally NOT done here — it
-- must ship together with the Webhooks.tsx narrowed-select to avoid 500ing the
-- admin page (tracked as a separate coordinated change).
-- ============================================================================

-- 1) Tenant + owner helpers (self-contained; bodies identical to live) --------
CREATE OR REPLACE FUNCTION public.current_user_digicom_business_id()
RETURNS uuid
LANGUAGE sql
STABLE SECURITY DEFINER
SET search_path TO 'public'
AS $$
  SELECT u."DigiCom Business"
  FROM public.users u
  WHERE u.id = auth.uid()
  LIMIT 1;
$$;
REVOKE ALL ON FUNCTION public.current_user_digicom_business_id() FROM PUBLIC;
GRANT EXECUTE ON FUNCTION public.current_user_digicom_business_id() TO authenticated, service_role;

CREATE OR REPLACE FUNCTION public.is_owner()
RETURNS boolean
LANGUAGE sql
STABLE SECURITY DEFINER
SET search_path TO 'public', 'pg_catalog'
AS $$
  SELECT EXISTS (
    SELECT 1 FROM public.users WHERE id = auth.uid() AND role = 'owner'
  );
$$;
REVOKE ALL ON FUNCTION public.is_owner() FROM PUBLIC;
GRANT EXECUTE ON FUNCTION public.is_owner() TO authenticated, service_role;

 RETURNS trigger LANGUAGE plpgsql SECURITY DEFINER SET search_path TO 'public', 'pg_catalog'

    IF NEW.status_id IS NOT NULL THEN

    IF v_is_delivered AND NEW.order_id IS NOT NULL THEN
        SELECT id INTO unpaid_status_id
        FROM public.order_statuses
        WHERE code ILIKE 'UNPAID' AND (active IS DISTINCT FROM false)
        LIMIT 1;

            UPDATE public.orders o

              AND o.current_courier_return_note_id IS NULL
              AND o.replace IS NOT TRUE;
        END IF;
    END IF;

    RETURN NEW;
END;

CREATE OR REPLACE FUNCTION public.set_courier_parcel_delivered_at()
 RETURNS trigger LANGUAGE plpgsql SECURITY DEFINER SET search_path TO 'public'
AS $function$
DECLARE
    v_is_delivered boolean := false;
BEGIN
    IF NEW.status_id IS NULL THEN
        RETURN NEW;
    END IF;

    IF TG_OP = 'UPDATE' AND NEW.status_id IS NOT DISTINCT FROM OLD.status_id THEN
        RETURN NEW;
    END IF;

    SELECT ps.is_delivered INTO v_is_delivered
    FROM public.package_statuses ps
    WHERE ps.id = NEW.status_id;

    IF COALESCE(v_is_delivered, false) AND NEW.delivered_at IS NULL THEN
        NEW.delivered_at := now();
    END IF;

    RETURN NEW;
END;
